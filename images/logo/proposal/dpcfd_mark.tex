% dpcfd logo -- variant: dpcfd_mark
\documentclass[tikz,border=0pt]{standalone}

% ---- colours ---------------------------------------------------------------
\definecolor{dpdisc}{HTML}{9CE2FF}        % disc (centre, when shaded)
\definecolor{dpdiscedge}{HTML}{9CE2FF}% disc edge (only used when shaded)
\definecolor{dpmarkin}{HTML}{000000}    % mark inside the disc
\definecolor{dpmarkout}{HTML}{000000}  % mark where it leaves the disc

% ---- parameters (fractions of the disc radius) -----------------------------
\newcommand{\logoscale}{3cm}            % disc radius on paper
\pgfmathsetmacro{\slant}{16.7}          % stem lean from vertical, degrees
\pgfmathsetmacro{\cdrop}{0.06}          % spiral centre below the disc centre
\pgfmathsetmacro{\rin}{0.17}            % spiral radius at the inner end
\pgfmathsetmacro{\pitch}{0.38}          % radius gained per turn
\pgfmathsetmacro{\turns}{2}             % number of turns
\pgfmathsetmacro{\rout}{\rin+\pitch*\turns}
\pgfmathsetmacro{\win}{0.12}         % stroke width at the inner end
\pgfmathsetmacro{\wout}{0.055}       % stroke width at the outer end
\pgfmathsetmacro{\handle}{0.34}         % ascender length above the spiral
\pgfmathsetmacro{\stemlen}{1.00}        % stem length below the centre
\pgfmathsetmacro{\barleft}{0.12}      % crossbar reach, left of the stem
\pgfmathsetmacro{\barright}{0.08}     % crossbar reach, right of the stem
\pgfmathsetmacro{\kk}{\pitch/(2*pi)}    % dr/dtheta of the spiral

% ---- spiral band: offset the centre line along its true normal -------------
% s runs from 0 (outer end) to \turns (inner end)
\pgfmathdeclarefunction{sr}{1}{\pgfmathparse{\rin+\pitch*(\turns-#1)}}
\pgfmathdeclarefunction{sw}{1}{\pgfmathparse{\wout+(\win-\wout)*#1/\turns}}
\pgfmathdeclarefunction{st}{1}{\pgfmathparse{90+360*#1}}
\pgfmathdeclarefunction{sn}{1}{\pgfmathparse{sqrt(sr(#1)*sr(#1)+\kk*\kk)}}
\pgfmathdeclarefunction{ex}{2}{\pgfmathparse{sr(#1)*cos(st(#1))+#2*sw(#1)/2*(sr(#1)*cos(st(#1))-\kk*sin(st(#1)))/sn(#1)}}
\pgfmathdeclarefunction{ey}{2}{\pgfmathparse{sr(#1)*sin(st(#1))+#2*sw(#1)/2*(sr(#1)*sin(st(#1))+\kk*cos(st(#1)))/sn(#1)}}

% the mark in one colour: spiral band, stem, ascender, crossbar
\newcommand{\dpmark}[1]{%
  \begin{scope}[shift={(0,-\cdrop)}, rotate=-\slant]
    \fill[#1]
      plot[domain=0:\turns, samples=721, variable=\s] ({ex(\s,1)},{ey(\s,1)})
      -- plot[domain=\turns:0, samples=721, variable=\s] ({ex(\s,-1)},{ey(\s,-1)})
      -- cycle;
    \draw[#1, line width=\win*\logoscale]  (0,-\stemlen) -- (0,\rin+\win/2);
    \draw[#1, line width=\wout*\logoscale] (0,\rout-\wout/2) -- (0,\rout+\handle);
    \draw[#1, line width=\win*\logoscale]  (-\barleft,0) -- (\barright,0);

  \end{scope}
}

\begin{document}
\begin{tikzpicture}[x=\logoscale, y=\logoscale, line cap=butt, line join=miter]
  \fill[dpdisc] (0,0) circle[radius=1];
  \dpmark{dpmarkin}
\end{tikzpicture}
\end{document}
